% ---- logo support ------------------------------------------
% Logos are IMAGES, which are ATS-safe: an embedded picture contributes nothing
% to the PDF text layer (unlike icon fonts). Put PNG/PDF files in
% assets/logos/ named exactly as referenced below. If a file is missing, the
% macros below fall back to plain text so the CV always builds.
\graphicspath{{assets/logos/}}

% \haslogo{name} is true only if assets/logos/<name>.png (or .pdf) exists.
% No raisebox here: the caller controls vertical alignment so it can be tuned
% per context (header vs. role line) without double-shifting.
\newcommand{\logoifexists}[2]{% #1 = file basename, #2 = height
  \IfFileExists{assets/logos/#1.png}{\includegraphics[height=#2]{#1.png}}{%
  \IfFileExists{assets/logos/#1.pdf}{\includegraphics[height=#2]{#1.pdf}}{}}%
}
% Vertically centre a logo on the current text line (aligns the logo's midline
% with the text midline, so logos read as sitting "next to" the words).
\newcommand{\vcenterlogo}[2]{\raisebox{-0.5\height+0.5ex}{\logoifexists{#1}{#2}}}

% Header link with logo + readable text.  \linklogo{url}{logo}{visible text}
\newcommand{\linklogo}[3]{\href{#1}{\logoifexists{#2}{9pt}\,#3}}

% Company logo shown next to a role title. \rolelogo{logo}{height}
% Symbol logos are sized for equal visual area (area = aspect * height^2,
% target ~170 pt^2): roche 9.6  anca 14.0  sanofi 12.8  cofidis 8.7.
% novartis is a wordmark, so it is sized to text cap-height (7.5pt) instead.
\newcommand{\rolelogo}[2]{\logoifexists{#1}{#2}}
